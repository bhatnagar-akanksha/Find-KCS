function filterDataByQuarters(selectedQuarters) {
  var filterDataByQuarter = Sheet1.filter(function (item) {
    var itemYear = item.createdOn.split("/")[2]; // Extract year from date
    var itemQuarter = getQuarter(item.createdOn);

    return selectedQuarters.some(function (selectedQuarter) {
      var [selectedYear, selectedQuarterOnly] = selectedQuarter.split("-");
      return itemYear === selectedYear && itemQuarter === selectedQuarterOnly;
    });
  });
  return filterDataByQuarter;
}

function findCommonObjectsanyTwoselected(array1, array2) {

  } else {
    // Combine both arrays into one and map each articleId to the number of occurrences
    const combined = [...array1, ...array2];

    const occurrenceMap = new Map();

    combined.forEach((item) => {
      if (occurrenceMap.has(item.articleId)) {
        occurrenceMap.set(
          item.articleId,
          occurrenceMap.get(item.articleId) + 1
        );
      } else {
        occurrenceMap.set(item.articleId, 1);
      }
    });

    // Filter to find articleIds that appear in both arrays
    const commonIds = Array.from(occurrenceMap)
      .filter(([key, value]) => value === 2) // Ensure the ID is in both arrays
      .map(([key, value]) => key);

    // Filter the original arrays to get the objects corresponding to the common IDs
    const commonObjects = combined.filter((item) =>
      commonIds.includes(item.articleId)
    );

    // Remove duplicates if needed (in case the same object is present multiple times)
    uniqueCommonObject = Array.from(
      new Map(commonObjects.map((item) => [item.articleId, item])).values()
    );
  }
  return uniqueCommonObject;
}

function getQuarter(dateString) {
  var parts = dateString.split("/");
  var month = parseInt(parts[0]);

  if (month == 1 || month == 2 || month == 12) {
    return "Q1";
  } else if (month >= 3 && month <= 5) {
    return "Q2";
  } else if (month >= 6 && month <= 8) {
    return "Q3";
  } else if (month >= 9 && month <= 11) {
    return "Q4";
  } else {
    return "Invalid month";
  }
}
